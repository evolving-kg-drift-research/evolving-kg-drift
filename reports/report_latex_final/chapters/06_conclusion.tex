\chapter{Kết luận và hướng phát triển}
\label{chap:conclusion}

\section{Tổng kết}
\label{sec:summary}
Đề tài muốn biết: khi đồ thị tri thức được cập nhật và mô hình học lại, các thực thể ``trôi'' đi bao nhiêu sau khi trừ phần nhiễu do học lại, và sự trôi đó có làm mô hình trả lời kém đi các câu hỏi nhiều bước không. Sau khi phát hiện kết quả phiên bản 28/10/2026 dựa trên dữ liệu sai ngày và mô hình chưa học xong, toàn bộ quy trình được làm lại, mỗi bước đều ghi lại cách làm trước khi chạy. Kết quả theo từng bước:
\begin{enumerate}
    \item \textbf{Dữ liệu:} đã sửa ngày tháng cho 10 lát cắt (27/04/2025--15/08/2026), sửa cách nhận diện thực thể, loại riêng các phần dữ liệu đáng ngờ kèm lý do.
    \item \textbf{Mô hình:} TransE-L2 đoán tốt hơn cách đoán đơn giản ở cả 10 lát cắt và chạy lại cho kết quả giống hệt; nhưng thực thể ít quan hệ lệch nhiều giữa các lần học lại.
    \item \textbf{Thước đo độ trôi:} trên toàn bộ thực thể, kết quả không khác gì ở nơi không có độ trôi; thước đo chỉ nhận ra thay đổi lớn ở nhóm thực thể có từ 3 quan hệ, và thang đo lệch dần khi học lâu hơn.
    \item \textbf{Giả thuyết $H_1$:} không kiểm định được theo quy tắc đã đặt trước; hệ số đổi dấu theo số vòng học, và dữ liệu chỉ đủ để phát hiện tác động rất lớn.
\end{enumerate}

\begin{remark}[Kết luận trong một câu]
Với đồ thị tri thức thưa như hiện tại, chưa thể biết độ trôi có làm mô hình suy luận kém đi hay không, vì nhiễu do học lại che lấp tín hiệu; đóng góp của đề tài là bộ quy trình kiểm chứng giúp phân biệt ``không có tác động'' với ``chưa đo được'', và chỉ ra cần thêm dữ liệu như thế nào.
\end{remark}

\section{Giới hạn}
\label{sec:limitations}
Bộ dữ liệu mới là bản ứng viên, chưa qua kiểm định chính thức của phần dữ liệu và chưa đối chiếu lại với Neo4j; độ chính xác khi trích quan hệ từ bài báo chưa được đo; nhận diện thực thể chưa hiểu ngữ cảnh; một số lựa chọn khi huấn luyện được sửa sau khi thấy kết quả (đã liệt kê đầy đủ); phép thử đối chứng dương chỉ dùng một kiểu thay đổi; giả thuyết $H_2$ (so sánh mô hình cố định và mô hình cập nhật) và $H_{3a}$ (suy luận theo đường đi, phân hệ M3) chưa được làm lại.

\section{Hướng phát triển}
\label{sec:future}
\begin{enumerate}
    \item \textbf{Thu thêm dữ liệu} để mỗi thực thể có nhiều quan hệ hơn (nhiều nguồn, nhiều bài về cùng một thực thể) và có nhiều chuyển tiếp hơn. Đây là cách duy nhất để tăng khả năng phát hiện.
    \item \textbf{Chỉnh thang đo $SED^+$} để không bị lệch theo số vòng học, ví dụ tính độ trôi và mức nhiễu với cùng số lần đo; ghi trước tiêu chí kiểm tra thang đo.
    \item \textbf{Đo chất lượng trích quan hệ và nhận diện thực thể} trên một bộ mẫu do người gán nhãn, rồi hoàn tất kiểm định chính thức phần dữ liệu và đối chiếu Neo4j.
    \item \textbf{Đa dạng hóa phép thử đối chứng dương} (thêm quan hệ, bớt quan hệ, đổi loại quan hệ) để biết thước đo nhạy đến đâu với những thay đổi giống thực tế hơn.
    \item Làm lại $H_2$ và $H_{3a}$, với cách làm được ghi trước, sau khi đáp ứng các điều kiện trên.
\end{enumerate}
