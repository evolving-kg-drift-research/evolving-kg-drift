\chapter{Kết quả thực nghiệm}
\label{chap:empirical_results}

Chương này trả lời từng câu hỏi theo đúng thứ tự của quy trình: dữ liệu có đúng không, mô hình có học tốt không, phép đo độ trôi có nhạy không, và cuối cùng độ trôi có ảnh hưởng tới khả năng suy luận không. Hình~\ref{fig:evidence_chain} tóm tắt câu trả lời cho cả năm bước. Toàn bộ kết quả thuộc giai đoạn thử nghiệm, chưa được nghiệm thu chính thức; mọi lựa chọn đều được ghi vào nhật ký quyết định \textit{trước} khi chạy.

\begin{remark}[Vì sao kết quả ngày 28/10/2026 bị rút lại]
Kết luận trước đây ($\beta_d = -0{,}00366$, $p = 0{,}574$, bộ 147 câu hỏi, so sánh mô hình cố định và mô hình cập nhật) không còn giá trị vì năm lỗi độc lập: (i) khoảng một nửa dữ liệu dùng ngày máy tải trang thay cho ngày đăng bài; (ii) mô hình chỉ học 50 vòng nên chưa học xong (báo cáo cũ còn ghi nhầm thuật toán là Adam trong khi chương trình thực tế dùng SGD); (iii) thực thể được nhận diện bằng cách so khớp chuỗi chữ, nên ``FPT'' và ``tập đoàn FPT'' là hai thực thể, còn chữ ``ông'' lại thành một thực thể nối với 17 thực thể khác; (iv) 121/147 câu hỏi chính là những quan hệ mô hình đã được học, nên chỉ kiểm tra trí nhớ; (v) đồ thị quá thưa để đo độ trôi của từng thực thể.
\end{remark}

\section{Bước 1 --- Dữ liệu có đúng thời gian không?}
\label{sec:m1_repair}

\subsection{Những gì sai trong dữ liệu cũ}
Gói dữ liệu cũ không bị hỏng tệp (mọi mã băm đều khớp) nhưng \textbf{sai về thời gian}:
\begin{itemize}
    \item \textbf{Ngày bị ghi nhầm:} 923/1.803 bản ghi lấy ngày máy tải trang (06--07/09/2026) làm ngày ``biết được tin''. Vì vậy ba lát cắt cuối chỉ dài $0{,}33$; $0{,}01$ và $0{,}01$ ngày --- chúng xếp theo thứ tự máy tải trang, không theo thời gian thật. Kiểm tra bằng mã số bài báo (mã bài VnExpress tăng dần theo thời gian, khớp gần tuyệt đối với ngày lưu trữ, hệ số tương quan hạng $0{,}999$) cho thấy các bài ``muộn'' thực ra được đăng rải rác từ 03/2025 đến 08/2026.
    \item \textbf{Ranh giới cắt sai:} 5/9 ranh giới giữa các lát cắt chia đôi những sự kiện cùng một thời điểm.
    \item \textbf{Nhận diện thực thể sai:} ít nhất 31 nhóm tên gọi khác nhau của cùng một thực thể; các từ chung chung như ``ông'', ``công ty'' bị coi là thực thể.
    \item \textbf{Tiêu đề tin liên quan:} 48 trang Tuổi Trẻ có phần nội dung chỉ là dòng tiêu đề tin liên quan ở rìa trang; ngày đăng của trang không cho biết khi nào dòng tiêu đề đó xuất hiện.
\end{itemize}

\subsection{Cách sửa}
Các quy tắc sau được duyệt và ghi lại trước khi áp dụng:
\begin{enumerate}
    \item \textbf{Ngày biết được tin} = ngày bản lưu trữ Wayback nếu có (880 bản ghi); nếu không có thì lấy ngày đăng mà tờ báo tự ghi trong trang (871 bản ghi, đọc từ phần dữ liệu cấu trúc JSON-LD). Ngày máy tải trang không bao giờ được dùng. Hai cách lấy ngày lệch nhau rất ít (trung vị $1{,}22$ ngày).
    \item Dòng tiêu đề tin liên quan không có bản lưu trữ bị loại riêng ra (45 sự kiện, $3{,}7\%$).
    \item Ranh giới lát cắt chia đều số sự kiện nhưng không bao giờ cắt ngang những sự kiện cùng thời điểm.
    \item \textbf{Nhận diện thực thể:} chuẩn hóa tên theo những quy tắc chỉ nhìn vào chính tên đó (bỏ ``tập đoàn'', ``công ty'', ``ông'', ``bà'' ở đầu; bỏ ``Corporation'', ``Inc'' ở cuối\ldots), nên không cần biết trước thông tin tương lai. Loại 22 từ chung chung và gộp 33 cách viết trùng; 1.389 còn 1.334 thực thể. Loại 40 sự kiện chứa từ chung chung ($3{,}3\%$).
\end{enumerate}

\section{Bước 2 --- Mô hình có học tốt không?}
\label{sec:backbone}

\subsection{Chọn cấu hình}
Mô hình TransE-L2 được thử với 12 cấu hình (kích thước biểu diễn 32 hoặc 64; tốc độ học $0{,}01$ hoặc $0{,}05$; ngưỡng phân tách $0{,}5$; $1{,}0$ hoặc $2{,}0$). Mỗi cấu hình được chấm điểm bằng cách giấu bớt một phần quan hệ rồi xem mô hình có đoán lại được không (chỉ số MRR, càng gần 1 càng tốt). Cấu hình tốt nhất: kích thước 64, tốc độ học $0{,}01$, ngưỡng phân tách $2{,}0$, MRR $0{,}35$.

\subsection{Học bao nhiêu vòng là đủ?}
Tiêu chí ``học xong'' đặt ra ban đầu có lỗi: một ngưỡng không thể đạt được với cấu hình đã chọn, một ngưỡng khác nhỏ hơn cả độ dao động tự nhiên của quá trình học. Năm cách khác được ghi lại trước rồi thử lần lượt, kể cả đổi sang thuật toán Adam (kết quả kém hơn). Không cách nào thỏa đúng tiêu chí đã đặt. Quyết định cuối cùng là \textbf{học cố định 400 vòng} cho mọi lát cắt, và bắt buộc mọi kết luận phải giữ nguyên khi đổi sang 300 hoặc 500 vòng (ký hiệu $E$ là số vòng huấn luyện).

\subsection{Chất lượng và độ ổn định}
Bảng~\ref{tab:quality}: ở cả 10 lát cắt, mô hình đoán tốt hơn hẳn cách đoán đơn giản ``chọn thực thể hay xuất hiện nhất với quan hệ đó'' (MRR $0{,}31$--$0{,}44$ so với $0{,}09$--$0{,}13$).

Bảng~\ref{tab:stability} và Hình~\ref{fig:noise_degree} đo \textbf{độ ổn định}: học lại ba lần trên cùng dữ liệu rồi xem vị trí của cùng một thực thể lệch nhau bao nhiêu (0 = trùng khít; khoảng 1 = lệch như hai điểm ngẫu nhiên). Thực thể càng có nhiều quan hệ thì càng ổn định: nhóm có 1--2 quan hệ lệch $0{,}21$--$0{,}55$, nhóm 3--5 quan hệ lệch $0{,}17$--$0{,}33$, nhóm từ 6 quan hệ lệch $0{,}12$--$0{,}15$.

\section{Bước 3 --- Phép đo độ trôi có nhạy không?}
\label{sec:population_drift}
Muốn biết một thước đo có dùng được không, cần hai phép thử:
\begin{itemize}
    \item \textbf{Phép thử đối chứng âm:} đo ở nơi \textit{chắc chắn không có} độ trôi (so hai lần học trên cùng dữ liệu). Thước đo tốt phải báo ``không có gì''.
    \item \textbf{Phép thử đối chứng dương:} cố ý thay đổi quan hệ của một số thực thể đã biết trước, học lại, rồi xem thước đo có nhận ra chính những thực thể đó không.
\end{itemize}
Trước khi đo, các không gian biểu diễn được xoay về cùng hệ trục bằng phép Procrustes; mỗi thực thể được đo bằng một phép xoay \textit{không dùng chính nó} (chia 5 phần, căn chỉnh chéo), để tránh tự so với chính mình. Mức nhiễu của mỗi thực thể lấy từ 6 lần so sánh giữa các lần học lại.

Bảng~\ref{tab:population}: độ trôi đo được trên toàn bộ thực thể \textbf{gần như giống hệt} kết quả ở nơi không có độ trôi. Bảng~\ref{tab:poscontrol_population}: khi cố ý thay toàn bộ quan hệ của 10\% thực thể, thước đo chỉ nhận ra $10$--$15\%$ trường hợp (tiêu chí đặt trước là ít nhất 50\%). Hình~\ref{fig:detect_degree} cho thấy lý do: với thực thể có 1--2 quan hệ, thước đo gần như mù; với thực thể từ 3 quan hệ, nó nhận ra phần lớn thay đổi.

\section{Bước 4 --- Thu hẹp vào nhóm thực thể có từ 3 quan hệ}
\label{sec:n1}
Vì thước đo chỉ có tác dụng với thực thể có nhiều quan hệ, nhóm này (khoảng 10\% thực thể) được chọn làm đối tượng chính, kèm tiêu chí riêng được ghi trước: thước đo phải nhận ra ít nhất 80\% thay đổi lớn và báo động giả không quá 10\%, \textbf{ở cả ba mức} 300, 400, 500 vòng. Mức nhiễu được tính riêng trong nhóm, với ít nhất 30 thực thể tham chiếu (con số này được chọn sau khi đã biết số thực thể của từng chuyển tiếp, để giữ được 7 chuyển tiếp thay vì 5).

Bảng~\ref{tab:poscontrol_s3} và Hình~\ref{fig:poscontrol}: với 149 thực thể bị thay đổi có chủ đích, thước đo nhận ra $81\%$, $99\%$, $99\%$ thay đổi lớn ở 300, 400, 500 vòng. Nhưng \textbf{tỷ lệ báo động giả tăng theo số vòng học} ($4{,}3\% \to 9{,}0\% \to 17{,}0\%$), nên tiêu chí không đạt ở 500 vòng. Ngay ở nơi không có độ trôi, tỷ lệ bị đánh dấu ``trôi mạnh'' cũng tăng từ $2{,}7\%$ lên $10{,}3\%$, trong khi lẽ ra chỉ khoảng $2{,}3\%$: \textit{thang đo bị lệch dần khi học lâu hơn}.

\section{Bước 5 --- Độ trôi có làm mô hình trả lời kém đi không?}
\label{sec:h1}

\subsection{Bộ câu hỏi}
Mỗi câu hỏi đi qua 2 hoặc 3 bước quan hệ, và thực thể ở giữa được giấu đi. Ví dụ: \textit{``Bắt đầu từ Tim Cook, đi theo chuỗi quan hệ: là giám đốc điều hành của $\to$ hợp tác với $\to$ đầu tư vào --- thực thể cuối là gì?''} (mắt xích bị giấu: Apple, Google; đáp án: Anthropic). Câu hỏi bị loại nếu đáp án nối thẳng với điểm xuất phát (câu hỏi quá dễ) hoặc nếu có thực thể nào trên đường đi có ít hơn 3 quan hệ. Mô hình trả lời bằng cách cộng các vector quan hệ rồi tìm thực thể gần nhất; điểm RR bằng 1 nếu đáp án đứng đầu, bằng $1/2$ nếu đứng thứ hai, và cứ thế. Phần chữ tiếng Việt của câu hỏi chỉ để con người đọc; 35 câu mẫu đang chờ duyệt tay.

Bảng~\ref{tab:panel} và Hình~\ref{fig:panel}: có 590 lượt đo (câu hỏi $\times$ chuyển tiếp) từ 259 câu hỏi, nhưng dồn gần hết vào các chuyển tiếp cuối. Chỉ những câu hỏi xuất hiện ở ít nhất hai chuyển tiếp mới giúp ích cho phép so sánh: còn \textbf{404 lượt đo, 127 câu hỏi, 7 chuyển tiếp}. Điểm RR trung bình chỉ khoảng $0{,}09$, nghĩa là mô hình hiếm khi đưa đúng đáp án lên đầu danh sách.

\subsection{Kết quả}
Mô hình TWFE so sánh \textit{cùng một câu hỏi} qua các chuyển tiếp: nếu ở chuyển tiếp nào các thực thể trên đường đi trôi nhiều hơn, thì điểm RR có thấp hơn không? Hệ số $\beta$ âm nghĩa là ``trôi nhiều thì trả lời kém đi''. Mức chắc chắn được tính bằng bootstrap theo cụm (9.999 lần lấy mẫu lại, trọng số Webb). Bảng~\ref{tab:h1} và Hình~\ref{fig:h1_forest}:
\begin{itemize}
    \item $\beta = +0{,}025$; $-0{,}242$; $-0{,}329$ ở 300, 400, 500 vòng (giá trị $p$ lần lượt $0{,}86$; $0{,}13$; $0{,}06$). \textbf{Dấu đổi theo số vòng học} và không kết quả nào đạt mức chắc chắn thường dùng ($p < 0{,}05$).
    \item \textbf{Dữ liệu chỉ phát hiện được tác động rất lớn:} để có 80\% cơ hội nhận ra một tác động, tác động đó phải làm điểm RR giảm khoảng $0{,}015$--$0{,}062$ cho mỗi mức trôi điển hình, tức $15$--$65\%$ điểm RR trung bình (chỉ số MDE$_{80}$).
    \item \textbf{Phép kiểm chứng bằng độ trôi giả} (thay độ trôi thật bằng độ trôi đo ở nơi không có thay đổi) ở 300 vòng lại cho $\beta = +0{,}49$, $p = 0{,}067$ --- gần như ``có kết quả'' dù biết chắc là không có, thêm một dấu hiệu thang đo chưa chuẩn.
    \item Một cách đo phụ (phần dư có dấu) cho hệ số âm ở cả ba mức ($p = 0{,}049$; $0{,}050$; $0{,}016$). Kết quả này \textbf{không được coi là phát hiện}: đã thử tới 24 mô hình nên dễ gặp may, chưa có phép kiểm chứng giả cho riêng cách đo này, và nó có thể chỉ phản ánh ``thực thể nào kém ổn định thì vừa lệch nhiều vừa bị trả lời sai'', chứ không phải độ trôi thật.
\end{itemize}
Theo quy tắc đã ghi trước --- phép thử đối chứng dương phải đạt ở cả ba mức số vòng, có ít nhất 100 lượt đo hữu ích, ít nhất 5 chuyển tiếp, và kết luận không đổi theo số vòng học --- \textbf{giả thuyết $H_1$ được xếp loại không kiểm định được}. Các hệ số ở trên chỉ để mô tả.

Kết quả không cho biết độ trôi có hại hay vô hại. Mọi thanh sai số đều cắt qua 0, hệ số còn đổi dấu khi học lâu hơn. Muốn trả lời được câu hỏi này, cần nhiều dữ liệu hơn nhiều --- không phải phân tích khéo hơn.
\end{remark}
