\chapter{Thảo luận}

\section{Vì sao không trả lời được câu hỏi chính}
\label{sec:why_not_testable}
Mọi kết quả ở Chương~\ref{chap:empirical_results} chỉ về một hướng: \textbf{tiếng ồn to hơn tín hiệu}. Có bốn lý do, xếp từ gốc rễ đến hệ quả.

\begin{enumerate}
    \item \textbf{Dữ liệu quá thưa.} Mỗi lát cắt chỉ có khoảng 120 sự kiện mới, và khoảng 90\% thực thể chỉ xuất hiện trong 1--2 quan hệ. Một thực thể như vậy giống một điểm chỉ được buộc bằng một sợi dây: mỗi lần học lại, nó có thể quay về bất kỳ hướng nào quanh sợi dây đó. Đổi thuật toán học không giải quyết được (thử Adam còn tệ hơn), vì vấn đề nằm ở dữ liệu.
    \item \textbf{Thước đo chỉ nhạy với thay đổi lớn, và chỉ ở thực thể có nhiều quan hệ.} Thay toàn bộ quan hệ của một thực thể ít quan hệ thì thước đo gần như không nhận ra; với thực thể từ 3 quan hệ thì nhận ra $81$--$99\%$. Trong thực tế, thay đổi tự nhiên của một thực thể thường nhỏ hơn nhiều mức ``thay toàn bộ'' này.
    \item \textbf{Thang đo bị lệch khi học lâu hơn.} Ở nơi không có độ trôi, tỷ lệ bị đánh dấu ``trôi mạnh'' tăng từ $2{,}7\%$ lên $10{,}3\%$ khi tăng từ 300 lên 500 vòng học. Một giải thích có thể (chưa kiểm chứng): độ trôi của mỗi thực thể lấy trung vị của 3 lần đo, còn mức nhiễu lấy trung vị của 6 lần đo, nên độ trôi tự nhiên dao động mạnh hơn mức nhiễu ngay cả khi không có thay đổi nào.
    \item \textbf{Quá ít dữ liệu để so sánh.} Chỉ có 404 lượt đo hữu ích từ 127 câu hỏi và 7 chuyển tiếp. Với cỡ mẫu này, chỉ những tác động làm điểm RR giảm $15$--$65\%$ mới có cơ hội tốt để được nhận ra. Ước lượng thô: muốn phát hiện được tác động nhỏ bằng một nửa, cần khoảng gấp bốn lần số lượt đo hữu ích.
\end{enumerate}

\begin{remark}[Điều quan trọng nhất của chương]
``Không tìm thấy tác động'' \textbf{khác hoàn toàn} với ``đã chứng minh không có tác động''. Muốn nói ``không có'', phải chứng minh được rằng nếu có thì phép đo đã nhận ra --- điều mà dữ liệu hiện tại không làm được. Kết luận phiên bản 28/10/2026 (``độ trôi độc lập với khả năng suy luận'', $p = 0{,}574$) mắc đúng lỗi này, cộng thêm lỗi ngày tháng và mô hình chưa học xong.
\end{remark}

\section{Những điểm có thể làm kết quả sai lệch}
\label{sec:threats}
\begin{itemize}
    \item \textbf{Từ bên trong nghiên cứu:} ngày đăng do tờ báo tự ghi có thể đã bị sửa; nhận diện thực thể chỉ dựa trên tên, không hiểu ngữ cảnh (không biết ``ông'' là ai); độ chính xác khi trích quan hệ từ bài báo chưa được đo --- một vài câu hỏi mẫu có quan hệ đáng ngờ; nhiều lựa chọn khi huấn luyện được sửa sau khi đã thấy kết quả (đã liệt kê đủ ở Bảng~\ref{tab:variants}).
    \item \textbf{Từ chính thước đo:} thang đo $SED^+$ lệch dần theo số vòng học; cách tạo thay đổi có chủ đích (thay quan hệ) chỉ là một kiểu và mạnh hơn thay đổi tự nhiên; mô hình chưa học xong hoàn toàn ở 400 vòng.
    \item \textbf{Khi áp dụng ra ngoài:} chỉ một lĩnh vực (tin công nghệ tiếng Việt từ hai tờ báo), 10 lát cắt, đồ thị nhỏ; không suy ra được cho đồ thị lớn và dày.
    \item \textbf{Về nghiệm thu:} bộ dữ liệu mới là bản ứng viên, chưa qua các cổng kiểm định chính thức của phần dữ liệu và chưa đối chiếu lại với Neo4j; giả thuyết $H_2$, $H_{3a}$ chưa làm lại.
\end{itemize}

\section{Đề tài được phép nói gì, không được nói gì}
\label{sec:defense}
\textbf{Được phép nói:}
\begin{enumerate}
    \item Đã phát hiện và sửa các lỗi về ngày tháng, nhận diện thực thể và bộ câu hỏi; mọi bước đều kiểm tra độc lập và chạy lại cho kết quả giống hệt.
    \item Với đồ thị tri thức thưa cỡ này, \textbf{nhiễu do học lại lớn hơn độ trôi} ở phần lớn thực thể; khả năng phát hiện độ trôi phụ thuộc mạnh vào số quan hệ của thực thể.
    \item Muốn kiểm định được $H_1$ cần: nhiều thực thể có từ 3 quan hệ trở lên, nhiều chuyển tiếp hơn, và một thang đo $SED^+$ không bị lệch theo số vòng học.
\end{enumerate}
\textbf{Không được phép nói:} độ trôi có làm hại hay không làm hại khả năng suy luận nhiều bước; và mọi kết luận về mô hình cố định so với mô hình cập nhật của phiên bản trước.
