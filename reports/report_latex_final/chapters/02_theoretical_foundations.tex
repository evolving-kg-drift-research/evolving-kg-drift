\chapter{Cơ sở lý thuyết và phương pháp toán học}
\label{chap:theoretical_foundations}
\label{chap:foundations}

Chương này giải thích cách đo độ trôi biểu diễn trên đồ thị tri thức thay đổi theo thời gian. Nội dung gồm cách biểu diễn tri thức bằng vector, đưa các không gian về cùng hệ tọa độ, ước lượng nhiễu nền và phân tích bằng kinh tế lượng. Các phương pháp đều nhằm giữ đúng trục thời gian, loại bỏ sự phụ thuộc thống kê gây sai lệch và bảo đảm độ tin cậy cao nhất cho suy luận.

\section{Mô hình thời gian và tính toàn vẹn thời gian}
\label{sec:temporal_model_integrity}

\subsection{Đồ thị tri thức thay đổi theo thời gian}
Đồ thị tri thức động là chuỗi trạng thái đồ thị thay đổi theo thời gian thực. Ở mỗi mốc rời rạc $t \in \{1, 2, \dots, T\}$, trạng thái đồ thị gồm ba thành phần:
\begin{equation}
\mathcal{G}_t = \left(\mathcal{E}_t, \mathcal{R}_t, \mathcal{T}_t\right)
\end{equation}
trong đó:
\begin{itemize}
    \item $\mathcal{E}_t \subset \mathcal{E}$ là tập hữu hạn các thực thể đã tồn tại và được xác định danh tính tại thời điểm $t$.
    \item $\mathcal{R}_t \subset \mathcal{R}$ là tập quan hệ có hiệu lực tại thời điểm $t$.
    \item $\mathcal{T}_t \subseteq \mathcal{E}_t \times \mathcal{R}_t \times \mathcal{E}_t$ là tập bộ ba (chủ thể, quan hệ, đối tượng) $\tau = (h, r, t)$. Trong đó, $h$ là chủ thể, $r$ là quan hệ và $t$ là đối tượng.
\end{itemize}

Đồ thị phát triển bằng cách thêm thực thể mới ($\mathcal{E}_{t-1} \subset \mathcal{E}_t$), cập nhật hoặc bãi bỏ quan hệ và sắp xếp lại các liên kết giữa các nút.

\subsection{Phân biệt ba trục thời gian}
Nhầm lẫn các loại thời gian là một nguyên nhân chính gây rò rỉ dữ liệu và tạo kết quả sai trong nghiên cứu đồ thị tri thức động. Vì vậy, đề tài tách rõ ba trục thời gian độc lập:

\begin{enumerate}
    \item \textbf{Thời gian có hiệu lực trong thực tế:}
    Khoảng $[\tau_{\text{valid\_from}}, \tau_{\text{valid\_to}}]$ cho biết một sự kiện hoặc quan hệ thực sự tồn tại và đúng trong thực tế từ khi nào đến khi nào.
    \item \textbf{Thời gian quan sát bằng chứng công khai:}
    Dấu thời gian $t_{\text{evidence}}$ (trường \texttt{evidence\_\allowbreak observed\_\allowbreak at}) cho biết khi nào phiên bản bằng chứng, bài báo hoặc tài liệu chứa bộ ba được công bố công khai lần đầu trên nguồn có thể kiểm tra độc lập. Trục thời gian này quyết định một sự thật có được đưa vào đồ thị tại mốc chốt hay không.
    \item \textbf{Thời gian hệ thống thu thập dữ liệu:}
    Dấu thời gian $t_{\text{crawl}} \equiv \texttt{retrieved\_at\_real}$ (hoặc $\texttt{ingested\_at\_real}$) ghi lại lúc trình thu thập tải và lưu bản ghi vào hệ thống.
\end{enumerate}

\begin{definition}[Không dùng trục thời gian này thay cho trục khác]
Ba trục thời gian không được hoán đổi hoặc suy ra để thay thế cho nhau:
\begin{equation}
[\tau_{\text{valid\_from}}, \tau_{\text{valid\_to}}] \neq t_{\text{evidence}} \neq t_{\text{crawl}}
\end{equation}
Một sự thật có thể đã đúng từ năm 2020 ($\tau_{\text{valid\_from}} = 2020$), nhưng báo chí chỉ công bố bằng chứng vào ngày 15/01/2024 ($t_{\text{evidence}} = \text{2024-01-15}$). Hệ thống có thể thu thập bài báo vào ngày 20/01/2024 ($t_{\text{crawl}} = \text{2024-01-20}$). Khi dựng đồ thị tại mốc chốt $T_{\text{cutoff}}$, phải dựa hoàn toàn vào $t_{\text{evidence}}$.
\end{definition}
Nói đơn giản, ngày sự việc xảy ra, ngày công chúng biết và ngày máy tải bài về là những mốc khác nhau.
\subsection{Các bất biến về thời gian}
Để phép đo độ trôi hợp lệ về khoa học và có thể được người khác tái lập, quy trình đặt ra các bất biến, tức những nguyên tắc không được vi phạm:

\begin{axiom}[Không dùng bằng chứng tương lai -- INV-01]
Mọi bộ ba $\tau \in \mathcal{T}_t$ trong lát cắt thời gian tại mốc chốt $T_{\text{cutoff}}$ phải thỏa mãn:
\begin{equation}
\texttt{evidence\_observed\_at}(\tau) \le T_{\text{cutoff}}
\end{equation}
Bằng chứng công bố sau $T_{\text{cutoff}}$ không được xuất hiện trong lát cắt thời gian, kể cả khi nó nói về sự kiện đã xảy ra trước đó.
\end{axiom}

\begin{axiom}[Không nhận diện thực thể bằng tri thức tương lai -- INV-02]
Bảng ánh xạ danh tính thực thể $M_t$ tại mốc chốt $T_{\text{cutoff}}$ chỉ được dùng các quy tắc liên kết danh tính đã có tại hoặc trước mốc này:
\begin{equation}
\texttt{mapping\_available\_at}(m) \le T_{\text{cutoff}}, \quad \forall m \in M_t
\end{equation}
Không dùng danh mục thực thể hoàn chỉnh của tương lai để gộp hoặc xác định danh tính các nút trong quá khứ.
\end{axiom}

\begin{axiom}[Nguồn gốc chỉ ghi thêm, không sửa xóa -- INV-03]
Kho phiên bản sự thật ($\texttt{FactVersion}$) và kho văn bản thô chỉ ghi thêm, không sửa xóa. Không được xóa ($\texttt{DELETE}$), cắt bỏ ($\texttt{TRUNCATE}$) hoặc sửa tại chỗ ($\texttt{UPDATE}$) dữ liệu lịch sử. Mỗi thay đổi phải tạo một phiên bản mới có đầy đủ thông tin nguồn gốc.
\end{axiom}

\begin{axiom}[Chạy lại cho kết quả giống hệt -- INV-04]
Khi dữ liệu đầu vào đã khóa, phiên bản mã nguồn, cấu hình và hạt giống ngẫu nhiên đều như nhau, việc dựng lại lát cắt thời gian và huấn luyện phải tạo ra các tệp trùng từng byte. Mã băm SHA-256 của các tệp phải giống hệt nhau:
\begin{equation}
\text{SHA-256}\left(\text{Pipeline}(\mathcal{D}, \text{Config}, \text{Seed})\right) \equiv \text{Hằng số}
\end{equation}
\end{axiom}

\section{Mô hình biểu diễn tri thức TransE-L2}
\label{sec:transe_l2_kge}

\subsection{Không gian biểu diễn hình học}
TransE \cite{bordes2013translating} biểu diễn thực thể và quan hệ bằng vector trong không gian Euclid thực có $d$ chiều:
\begin{equation}
\mathbf{h}, \mathbf{t} \in \mathbb{R}^d, \quad \mathbf{r} \in \mathbb{R}^d
\end{equation}
Nghiên cứu cố định số chiều tại $d = 64$ dựa trên khảo sát thử nghiệm. Lựa chọn này nhằm cân bằng tốt nhất giữa khả năng biểu diễn cấu trúc đồ thị và độ ổn định khi căn chỉnh trực giao.

\subsection{Quan hệ như phép dịch chuyển vector và khoảng cách L2}
TransE xem quan hệ $r$ như một phép dịch chuyển từ vector chủ thể $h$ đến vector đối tượng $t$:
\begin{equation}
\mathbf{h} + \mathbf{r} \approx \mathbf{t}
\end{equation}
Mức phù hợp của bộ ba ứng viên $\tau = (h, r, t)$ được đo bằng khoảng cách Euclid $L_2$, gọi là hàm năng lượng:
\begin{equation}
E(h, r, t) = \|\mathbf{h} + \mathbf{r} - \mathbf{t}\|_2 = \sqrt{\sum_{k=1}^d \left(h_k + r_k - t_k\right)^2}
\end{equation}
Bộ ba đúng có năng lượng $E(h, r, t)$ gần 0; bộ ba sai hoặc không tồn tại có năng lượng lớn.
Nói đơn giản, cộng vector quan hệ vào vector chủ thể rồi xem điểm đến gần vector đối tượng đến đâu.
\subsection{Hàm mất mát xếp hạng theo biên và cách tối ưu}
Nghiên cứu huấn luyện TransE bằng hàm mất mát xếp hạng theo biên, có thêm phần điều chuẩn để kiểm soát trọng số:
\begin{equation}
\mathcal{L} = \sum_{(h,r,t) \in \mathcal{T}^+} \sum_{(h',r,t') \in \mathcal{T}^-} \max\left(0, \gamma + E(h,r,t) - E(h',r,t')\right) + \lambda \sum_{e \in \mathcal{E}} \|\mathbf{e}\|_2^2
\label{eq:margin_ranking_loss}
\end{equation}
trong đó:
\begin{itemize}
    \item $\mathcal{T}^+$ là tập bộ ba đúng, gọi là mẫu dương, trong lát cắt thời gian $\mathcal{G}_t$.
    \item $\mathcal{T}^-$ là tập mẫu âm, tạo bằng cách thay chủ thể hoặc đối tượng và loại những bộ ba vốn đã có trong tập mẫu dương:
    \begin{equation}
    \mathcal{T}^- = \{(h', r, t) \mid h' \in \mathcal{E}, (h', r, t) \notin \mathcal{T}^+\} \cup \{(h, r, t') \mid t' \in \mathcal{E}, (h, r, t') \notin \mathcal{T}^+\}
    \end{equation}
    \item $\gamma > 0$ là biên phân cách; trong phiên bản tái thực hiện $\gamma = 2{,}0$, chọn bằng tìm kiếm lưới trên các bộ ba giữ lại để kiểm tra.
    \item $\lambda$ là trọng số điều chuẩn, dùng để kiểm soát độ lớn của vector.
\end{itemize}
Hàm mất mát phạt khi mẫu dương chưa có năng lượng thấp hơn mẫu âm đủ một biên; phần điều chuẩn phạt vector có độ lớn quá lớn.
\begin{algorithm}[H]
\caption{Tạo bộ ba sai để luyện mô hình, có lọc để không lấy nhầm bộ ba đúng}
\label{alg:negative_sampling}
\begin{algorithmic}[1]
\Require Bộ ba dương $(h, r, t) \in \mathcal{T}^+$, danh mục thực thể $\mathcal{E}$, tập sự thật $\mathcal{T}^+$, số lần thử tối đa $K=100$.
\State Tung đồng xu ngẫu nhiên Bernoulli $p \sim \text{Bernoulli}(0{,}5)$.
\For{$k = 1 \to K$}
    \If{$p == 1$} \Comment{Thay thế thực thể đầu}
        \State Chọn ngẫu nhiên $h' \sim \text{Uniform}(\mathcal{E})$.
        \If{$(h', r, t) \notin \mathcal{T}^+$} \Return $(h', r, t)$ \EndIf
    \Else \Comment{Thay thế thực thể cuối}
        \State Chọn ngẫu nhiên $t' \sim \text{Uniform}(\mathcal{E})$.
        \If{$(h, r, t') \notin \mathcal{T}^+$} \Return $(h, r, t')$ \EndIf
    \EndIf
\EndFor
\State \Return Bỏ qua mẫu nếu vượt quá $K$ lần thử (không tạo mẫu sai, ghi nhận và bỏ qua).
\end{algorithmic}
\end{algorithm}

Mô hình được tối ưu bằng SGD, cập nhật sau từng bộ ba (không gộp lô), với số chiều $d = 64$, tốc độ học $\eta = 0{,}01$, biên $\gamma = 2{,}0$ và $E = 400$ vòng huấn luyện cố định (kiểm tra độ nhạy thêm ở $E = 300$ và $E = 500$). Sau mỗi lần cập nhật trọng số, vector thực thể được chiếu về mặt cầu đơn vị để ngăn mô hình giảm mất mát chỉ bằng cách tăng độ dài vector lên vô hạn:
\begin{equation}
\mathbf{e} \leftarrow \frac{\mathbf{e}}{\max\left(1, \|\mathbf{e}\|_2\right)}, \quad \forall e \in \mathcal{E}
\end{equation}
Phép cập nhật trên chia vector cho độ dài của nó khi độ dài vượt giới hạn đơn vị.

\section{Căn chỉnh không gian bằng Procrustes trực giao có dịch tâm và SVD}
\label{sec:procrustes_alignment}

\subsection{Vì sao các lần huấn luyện có hệ tọa độ khác nhau}
Hàm mất mát \eqref{eq:margin_ranking_loss} ràng buộc khoảng cách tương đối giữa các vector. Tuy nhiên, xoay hoặc phản xạ cả không gian Euclid không làm thay đổi các khoảng cách đó. Vì vậy, hai lần huấn luyện độc lập có cùng cấu hình nhưng khác hạt giống ngẫu nhiên, hoặc dùng hai lát cắt thời gian liên tiếp, sẽ tạo ra các hệ tọa độ khác nhau:
\begin{equation}
\mathbf{e}^{(1)} \neq \mathbf{e}^{(2)} \quad \text{dù cấu trúc đồ thị không đổi.}
\end{equation}
Do đó, không thể so sánh trực tiếp vector thô rồi kết luận về thay đổi ý nghĩa. Trước hết phải đưa chúng về cùng hệ tọa độ bằng một phép biến đổi giữ nguyên khoảng cách. Bài toán Procrustes trực giao cho phép tìm phép biến đổi này~\cite{schonemann1966generalized}.

\subsection{Bài toán Procrustes trực giao có dịch tâm}
Giả sử cần đưa không gian nguồn $S$, có ma trận vector $X$, về không gian đích $T$, có ma trận vector $Y$. Để tránh bị chi phối bởi thực thể thay đổi cấu trúc mạnh, chỉ dùng các thực thể neo cố định để ước lượng phép căn chỉnh:
\begin{equation}
\mathcal{A}_{\text{fit}} \subset \mathcal{E}_S \cap \mathcal{E}_T, \quad |\mathcal{A}_{\text{fit}}| = m_{\text{fit}} = 101
\end{equation}
Tập thực thể neo này được chọn độc lập từ trước. Phân bố của nó được kiểm tra tách biệt với phần thực thể neo giữ lại để kiểm tra ($m_{\text{holdout}} = 66$).

Gọi $X_{\text{fit}} \in \mathbb{R}^{m \times d}$ và $Y_{\text{fit}} \in \mathbb{R}^{m \times d}$ là các ma trận chứa vector của thực thể neo ở không gian nguồn và đích. Tâm của mỗi tập được tính bằng trung bình các vector:
\begin{equation}
\boldsymbol{\mu}_s = \frac{1}{m} \sum_{i=1}^m \mathbf{x}_i \in \mathbb{R}^d, \quad \boldsymbol{\mu}_t = \frac{1}{m} \sum_{i=1}^m \mathbf{y}_i \in \mathbb{R}^d
\end{equation}
Trừ tâm tương ứng khỏi từng vector để có các ma trận đã dịch tâm:
\begin{equation}
\widetilde{X} = X_{\text{fit}} - \mathbf{1}_m \boldsymbol{\mu}_s^T, \quad \widetilde{Y} = Y_{\text{fit}} - \mathbf{1}_m \boldsymbol{\mu}_t^T
\end{equation}
Procrustes trực giao có dịch tâm tìm ma trận tối ưu $Q \in \mathcal{O}(d)$ để hai tập điểm sau dịch tâm gần nhau nhất. Sai lệch được đo bằng tổng bình phương các phần tử, tức bình phương chuẩn Frobenius:
\begin{equation}
\min_{Q \in \mathbb{R}^{d \times d}} \left\| \widetilde{X} Q - \widetilde{Y} \right\|_F^2 \quad \text{sao cho} \quad Q^T Q = I_d
\label{eq:procrustes_opt}
\end{equation}
Nói đơn giản, dùng các thực thể neo làm điểm mốc, đưa tâm về cùng chỗ rồi xoay hoặc phản xạ để hai đám mây điểm khớp nhau nhất.
\subsection{Tìm nghiệm bằng phân tích giá trị kỳ dị SVD}
Khai triển bình phương chuẩn Frobenius trong hàm mục tiêu \eqref{eq:procrustes_opt}:
\begin{align}
\left\| \widetilde{X} Q - \widetilde{Y} \right\|_F^2 &= \text{Tr}\left( (\widetilde{X} Q - \widetilde{Y})^T (\widetilde{X} Q - \widetilde{Y}) \right) \nonumber \\
&= \text{Tr}\left( Q^T \widetilde{X}^T \widetilde{X} Q \right) - 2 \text{Tr}\left( Q^T \widetilde{X}^T \widetilde{Y} \right) + \text{Tr}\left( \widetilde{Y}^T \widetilde{Y} \right) \nonumber \\
&= \text{Tr}\left( \widetilde{X}^T \widetilde{X} \right) + \text{Tr}\left( \widetilde{Y}^T \widetilde{Y} \right) - 2 \text{Tr}\left( Q^T C \right)
\end{align}
Ở đây, $C = \widetilde{X}^T \widetilde{Y} \in \mathbb{R}^{d \times d}$ là ma trận hiệp phương sai chéo, mô tả mối liên hệ giữa hai tập vector nguồn và đích.

Các phần $\text{Tr}(\widetilde{X}^T \widetilde{X})$ và $\text{Tr}(\widetilde{Y}^T \widetilde{Y})$ không đổi theo $Q$. Vì vậy, muốn sai lệch nhỏ nhất thì chỉ cần làm vết của tích ma trận còn lại lớn nhất, tức tổng các phần tử trên đường chéo lớn nhất:
\begin{equation}
\max_{Q \in \mathcal{O}(d)} \text{Tr}\left( Q^T C \right)
\end{equation}
Dùng phân tích giá trị kỳ dị SVD để tách ma trận $C$ thành:
\begin{equation}
C = U \Sigma V^T
\end{equation}
Trong đó, $U, V \in \mathcal{O}(d)$ là các ma trận trực giao; $\Sigma = \text{diag}(\sigma_1, \sigma_2, \dots, \sigma_d)$ chứa các giá trị kỳ dị trên đường chéo, xếp theo thứ tự $\sigma_1 \ge \sigma_2 \ge \dots \ge \sigma_d \ge 0$.

Đặt $Z = V^T Q^T U \in \mathcal{O}(d)$. Khi đó, phần cần cực đại hóa trở thành tổng các phần tử đường chéo nhân với giá trị kỳ dị tương ứng:
\begin{equation}
\text{Tr}\left( Q^T C \right) = \text{Tr}\left( Q^T U \Sigma V^T \right) = \text{Tr}\left( V^T Q^T U \Sigma \right) = \text{Tr}\left( Z \Sigma \right) = \sum_{k=1}^d z_{kk} \sigma_k
\end{equation}
Vì $Z$ trực giao, các phần tử đường chéo thỏa mãn $|z_{kk}| \le 1$. Tổng vết đạt cực đại khi và chỉ khi $z_{kk} = 1, \forall k$, tức $Z = I_d$. Từ đó có nghiệm:
\begin{equation}
V^T Q^T U = I_d \implies Q^T = V U^T \implies Q^* = U V^T
\label{eq:procrustes_solution}
\end{equation}
Công thức nghiệm ghép hai ma trận trực giao lấy từ SVD để tạo ra phép căn chỉnh tối ưu.
\begin{remark}[Cho phép cả phép phản xạ]
Trong không gian $d = 64$, nghiệm $Q^* = U V^T$ thuộc nhóm trực giao $\mathcal{O}(d)$, với $\det(Q^*) \in \{+1, -1\}$. Như vậy, nghiệm cho phép cả xoay và phản xạ. Hệ thống kiểm tra tính trực giao và hạng của ma trận bằng:
\begin{equation}
E_{\text{orth}} = \frac{1}{d} \| Q^T Q - I_d \|_F \le 10^{-6}, \qquad \frac{\text{rank}(\widetilde{X}_{\text{fit}})}{\min(d,\, m_{\text{fit}} - 1)} \ge 0{,}90
\end{equation}
Đồng thời, hệ thống so khoảng cách sau căn chỉnh trên phần thực thể neo giữ lại (không dùng để tìm nghiệm) với phần dùng để khớp:
\begin{equation}
\text{Gap} = \operatorname{median}_{\text{holdout}} \left\| \tilde{\mathbf{x}}_e Q^* - \tilde{\mathbf{y}}_e \right\|_2 - \operatorname{median}_{\text{fit}} \left\| \tilde{\mathbf{x}}_e Q^* - \tilde{\mathbf{y}}_e \right\|_2 \le 0{,}65
\end{equation}
\end{remark}
Sai số trực giao cho biết phép biến đổi có giữ được hình học hay không. Sai số trên phần giữ lại cho biết phép căn chỉnh có phù hợp cả với những thực thể neo không dùng để tìm nghiệm hay không.
\subsection{So sánh cô-sin sau biến đổi afin và cô-sin sau dịch tâm}
Để đo độ dịch chuyển của thực thể $e$ giữa hai không gian, có thể dùng hai cách đo mức khác nhau về hướng:

\begin{definition}[Cô-sin sau biến đổi afin -- Công thức 14 \& 26 của đề cương gốc]
Đưa vector nguồn $\mathbf{x} \in \mathbb{R}^d$ sang hệ tọa độ đích bằng cách trừ tâm nguồn, biến đổi trực giao rồi cộng tâm đích:
\begin{equation}
\mathbf{x}' = (\mathbf{x} - \boldsymbol{\mu}_s) Q^* + \boldsymbol{\mu}_t
\label{eq:affine_transform}
\end{equation}
Sau đó, đo mức khác hướng giữa vector nguồn đã biến đổi $\mathbf{x}'$ và vector đích $\mathbf{y}$ bằng:
\begin{equation}
d_{\text{aff}}(\mathbf{x}, \mathbf{y}) = 1 - \cos(\mathbf{x}', \mathbf{y}) = 1 - \frac{{\mathbf{x}'}^T \mathbf{y}}{\|\mathbf{x}'\|_2 \|\mathbf{y}\|_2}
\label{eq:affine_cosine}
\end{equation}
\end{definition}
Phép đo lấy một trừ đi độ tương đồng cô-sin: hai hướng càng giống nhau thì giá trị càng nhỏ.
\begin{definition}[Cô-sin sau dịch tâm]
Cách đo này chỉ xét thay đổi góc sau khi đã trừ toàn bộ độ lệch tâm của hai tập điểm:
\begin{equation}
\mathbf{x}_c = (\mathbf{x} - \boldsymbol{\mu}_s) Q^*, \quad \mathbf{y}_c = \mathbf{y} - \boldsymbol{\mu}_t
\end{equation}
\begin{equation}
d_{\text{cent}}(\mathbf{x}, \mathbf{y}) = 1 - \cos(\mathbf{x}_c, \mathbf{y}_c) = 1 - \frac{\mathbf{x}_c^T \mathbf{y}_c}{\|\mathbf{x}_c\|_2 \|\mathbf{y}_c\|_2}
\label{eq:centered_cosine}
\end{equation}
\end{definition}
Khác với cách trên, cả hai vector ở đây đều được so sánh quanh tâm của tập điểm, không cộng tâm đích trở lại.
\subsubsection{Khác biệt giữa hai cách đo}
Hai cách đo khác nhau ở việc xử lý sự dịch chuyển của toàn bộ đám mây điểm:
\begin{enumerate}
    \item \textbf{Cô-sin sau dịch tâm:} Không đổi khi toàn bộ không gian cùng tịnh tiến. Sau khi trừ các tâm $\boldsymbol{\mu}_s$ và $\boldsymbol{\mu}_t$, phép đo chỉ thấy thay đổi hướng quanh tâm. Tuy nhiên, trong TransE, vị trí tâm phản ánh mật độ liên kết chung của đồ thị. Loại bỏ hoàn toàn tọa độ tuyệt đối sẽ làm mất thông tin về sự dịch chuyển của các cụm thực thể trong không gian vector.
    \item \textbf{Cô-sin sau biến đổi afin theo đề cương:} Ánh xạ \eqref{eq:affine_transform} đưa tâm nguồn về gốc tọa độ, xoay trực giao bằng $Q^*$ rồi cộng $+\boldsymbol{\mu}_t$ để đưa về tâm đích. Khi đó, vector $\mathbf{x}'$ nằm trong hệ tọa độ đích. So sánh cô-sin giữa $\mathbf{x}'$ và $\mathbf{y}$ phản ánh chính xác độ trôi vị trí của thực thể trong hệ này, đồng thời giữ hình học tịnh tiến của TransE.
\end{enumerate}

\begin{example}[Ví dụ cho thấy hai cách đo khác nhau]
Xét không gian 2 chiều, với phép biến đổi đồng nhất $Q = I_2$.
Vector nguồn là $\mathbf{x} = (1, 0)^T$, tâm nguồn là $\boldsymbol{\mu}_s = (0, 0)^T$.
Vector đích là $\mathbf{y} = (10, 11)^T$, tâm đích là $\boldsymbol{\mu}_t = (10, 10)^T$.
Ta tính được:
\begin{itemize}
    \item Với cô-sin sau dịch tâm: $\mathbf{x}_c = (1, 0)^T$, $\mathbf{y}_c = (10-10, 11-10)^T = (0, 1)^T$.
    Khoảng cách sau dịch tâm là:
    \begin{equation}
    d_{\text{cent}}(\mathbf{x}, \mathbf{y}) = 1 - \frac{1 \cdot 0 + 0 \cdot 1}{1 \cdot 1} = 1 - 0 = 1{,}0000
    \end{equation}
    \item Với cô-sin sau biến đổi afin: $\mathbf{x}' = (1-0, 0-0)^T I_2 + (10, 10)^T = (11, 10)^T$.
    Khoảng cách sau biến đổi afin là:
    \begin{equation}
    d_{\text{aff}}(\mathbf{x}, \mathbf{y}) = 1 - \frac{11 \cdot 10 + 10 \cdot 11}{\sqrt{11^2 + 10^2} \sqrt{10^2 + 11^2}} = 1 - \frac{220}{221} \approx 0{,}004525
    \end{equation}
\end{itemize}
Ví dụ cho thấy hai cách đo cho kết quả rất khác. Cụm điểm cùng dịch chuyển với tâm, nên thay đổi thật nhìn trên toàn không gian là rất nhỏ. Cô-sin sau dịch tâm lại báo sai lệch cực đại ($1{,}0$), còn cô-sin sau biến đổi afin phản ánh đúng sự tương đồng cấu trúc ($0{,}0045$).
\end{example}

\section{Phân phối nhiễu nền và cách loại thực thể đang xét khỏi phần tham chiếu}
\label{sec:empirical_null_lto}

\subsection{Nhiễu ngẫu nhiên do học lại}
Khi học lại mô hình biểu diễn vector trên cùng dữ liệu hoặc trên lát cắt thời gian mới, vector của thực thể luôn dịch chuyển một khoảng do:
\begin{itemize}
    \item Trọng số ban đầu được chọn ngẫu nhiên.
    \item Thứ tự các bộ ba được xáo trộn ngẫu nhiên qua từng vòng huấn luyện bằng SGD.
    \item Các bộ ba làm mẫu âm được lấy ngẫu nhiên.
\end{itemize}
Không thể loại bỏ hoàn toàn nhiễu ngẫu nhiên này. Vì vậy, độ dịch chuyển theo thời gian $\delta_{time} > 0$ chưa có nghĩa là ý nghĩa của thực thể đã thay đổi. Muốn xác định độ trôi thật, phải so sánh khoảng cách đó với phân phối nhiễu nền đo được từ thực nghiệm.

\subsection{Thiết kế 6 cặp hạt giống ngẫu nhiên trên cùng lát cắt thời gian}
Nghiên cứu cố định ba hạt giống ngẫu nhiên độc lập: $\mathcal{S} = \{13, 37, 101\}$.
Ở mỗi lát cắt thời gian tại $t$, huấn luyện song song ba mô hình tương ứng với ba hạt giống ngẫu nhiên này.
Từ 3 hạt giống ngẫu nhiên, tạo được 3 cặp mô hình độc lập không phân biệt thứ tự:
\begin{equation}
\mathcal{P} = \left\{ (13, 37), (13, 101), (37, 101) \right\}
\end{equation}
Để ước lượng nhiễu nền cho chuyển tiếp từ $T_{t-1}$ sang $T_t$, căn chỉnh Procrustes rồi đo khoảng cách của từng thực thể $e$ giữa các cặp mô hình trên cả 2 lát cắt thời gian ở hai đầu:
\begin{equation}
\mathcal{D}_{\text{noise}}(e) = \bigcup_{s \in \{T_{t-1}, T_t\}} \bigcup_{(a, b) \in \mathcal{P}} \left\{ d\left(\mathbf{e}_{s, a}, \mathbf{e}_{s, b}\right) \right\}
\end{equation}
Như vậy, mỗi thực thể $e$ có đúng $2 \times 3 = 6$ quan sát nhiễu thô độc lập trong mỗi chuyển tiếp. Những quan sát này chỉ do tính ngẫu nhiên khi học lại trên đồ thị không đổi.

\subsection{Loại thực thể đang xét khỏi phần tham chiếu để tránh tự phụ thuộc}
Nếu dùng chính dữ liệu của đối tượng đang xét để lập phân phối kiểm định cho nó, phép kiểm sẽ bị phụ thuộc vào đối tượng đó và kém nhạy hơn.

\begin{definition}[Đơn vị thống kê trong phần tham chiếu không chứa thực thể đang xét]
Để đánh giá thực thể $e$, lập tập nhiễu nền tham chiếu $\mathcal{R}_{-e}$ từ các thực thể khác $u \neq e$ trong cùng nhóm cấu trúc. Mỗi thực thể tham chiếu $u$ chỉ đóng góp một giá trị: trung vị của 6 quan sát nhiễu của nó:
\begin{equation}
n(u) = \text{median}\left( \mathcal{D}_{\text{noise}}(u) \right) = \text{median}\left( \{d_{u, 1}, d_{u, 2}, \dots, d_{u, 6}\} \right)
\label{eq:entity_null_median}
\end{equation}
Tập tham chiếu sau khi loại chính thực thể đang xét được viết như sau:
\begin{equation}
\mathcal{R}_{-e} = \left\{ n(u) \mid u \in \mathcal{B}(e) \setminus \{e\} \right\}
\label{eq:lto_reference_set}
\end{equation}
Trong đó, $\mathcal{B}(e)$ là nhóm bậc liên kết chứa thực thể $e$.
\end{definition}
Nói đơn giản, muốn biết một thực thể dao động nhiều hay ít thì so nó với các thực thể tương tự, không để chính nó tham gia đặt mức chuẩn.
\begin{axiom}[Số mẫu tham chiếu tối thiểu]
Để phân phối kiểm định có đủ ý nghĩa thống kê, số thực thể tham chiếu sau khi loại mục tiêu phải thỏa mãn:
\begin{equation}
N_{\text{ref}} = |\mathcal{R}_{-e}| \ge 50
\end{equation}
Nếu $N_{\text{ref}} < 50$, phân phối nhiễu không hợp lệ. Khi đó, độ trôi đã hiệu chuẩn phải được ghi là không khả dụng ($\texttt{available = False}$), kèm lý do cụ thể. Không được gán mặc định bằng 0.
\end{axiom}

\subsection{Chia nhóm theo bậc liên kết và gộp nhóm liền kề}
Vector của thực thể có ít liên kết thường chịu biến động ngẫu nhiên do khởi tạo lớn hơn thực thể trung tâm có nhiều liên kết. Mức biến động này tỷ lệ nghịch rõ rệt với bậc liên kết. Vì vậy, phải ước lượng nhiễu nền riêng theo bậc của thực thể. Bậc là số bộ ba có thực thể làm chủ thể hoặc đối tượng:
\begin{equation}
\text{deg}(e) = \left| \{ (u, r, v) \in \mathcal{T} \mid u = e \lor v = e \} \right|
\end{equation}
Chia thực thể thành 3 nhóm theo ngưỡng bậc:
\begin{itemize}
    \item Nhóm bậc thấp: $\text{deg}(e) < B_1$.
    \item Nhóm bậc trung bình: $B_1 \le \text{deg}(e) < B_2$.
    \item Nhóm bậc cao: $\text{deg}(e) \ge B_2$.
\end{itemize}

Nếu một nhóm quá ít thực thể ($N_k < 51$, không bảo đảm còn ít nhất 50 sau khi loại mục tiêu), thuật toán sẽ gộp các nhóm liền kề theo quy tắc cố định, chạy lại cho kết quả giống hệt:

\begin{algorithm}[H]
\caption{Gộp nhóm bậc liền kề khi một nhóm có quá ít thực thể}
\label{alg:bucket_merging}
\begin{algorithmic}[1]
\Require Danh sách nhóm phân tầng $\mathcal{G} = [G_1, G_2, \dots, G_K]$, ngưỡng hỗ trợ tối thiểu $M=51$.
\While{$|\mathcal{G}| > 1$}
    \State Tìm nhóm thưa đầu tiên: $i = \min \{j \mid 0 < |G_j| < M\}$.
    \If{Không tồn tại nhóm thưa $i$} \State \textbf{break} \EndIf
    \State Xác định tập chỉ số lân cận hợp lệ: $\mathcal{N} = \{j \in \{i-1, i+1\} \mid 1 \le j \le |\mathcal{G}|\}$.
    \State Chọn nhóm lân cận có quy mô lớn nhất (ưu tiên chỉ số nhỏ nếu hòa):
    \State $p = \arg\max_{j \in \mathcal{N}} (|G_j|, -j)$.
    \State Hợp nhất hai nhóm liền kề: $G_{\text{new}} = G_{\min(i, p)} \cup G_{\max(i, p)}$.
    \State Cập nhật lại danh sách $\mathcal{G}$ bằng cách thay thế hai nhóm cũ bằng $G_{\text{new}}$.
    \State Ghi nhật ký biến đổi cấu trúc (audit log) phục vụ truy vết provenance.
\EndWhile
\State \Return Danh sách nhóm phân tầng sau hợp nhất $\mathcal{G}$.
\end{algorithmic}
\end{algorithm}

\section{Độ trôi biểu diễn đã chuẩn hóa}
\label{sec:sed_plus_drift}

\subsection{Độ dịch chuyển theo thời gian (\texorpdfstring{$\delta_{\text{time}}$}{delta\_time})}
Với chuyển tiếp từ $T_{t-1}$ sang $T_t$, độ dịch chuyển theo thời gian của thực thể $e$ là trung vị của 3 khoảng cách giữa các lần huấn luyện song song dùng cùng hạt giống ngẫu nhiên ở hai đầu:
\begin{equation}
\delta_{\text{time}}(e, t) = \text{median}\left( \left\{ d\left(\mathbf{e}_{T_{t-1}, s}, \mathbf{e}_{T_t, s}\right) \right\}_{s \in \{13, 37, 101\}} \right)
\end{equation}
Lấy trung vị qua 3 hạt giống ngẫu nhiên độc lập giúp loại ảnh hưởng của giá trị ngoại lai do một lần khởi tạo may rủi.

\subsection{Phần dư có dấu}
Để biết độ dịch chuyển theo thời gian có vượt mức biến động nền hay không, lấy độ dịch chuyển trừ trung vị nhiễu tham chiếu:
\begin{equation}
r(e, t) = \delta_{\text{time}}(e, t) - \text{median}\left(\mathcal{R}_{-e}\right)
\label{eq:signed_excess}
\end{equation}
Trong đó, $\mathcal{R}_{-e}$ là tập nhiễu tham chiếu đã loại thực thể đang xét, được định nghĩa tại \eqref{eq:lto_reference_set}.
\begin{itemize}
    \item $r(e, t) > 0$: Thực thể dịch chuyển xa hơn mức trung bình của các thực thể cùng nhóm bậc khi đồ thị không đổi $\implies$ Có thể có tín hiệu độ trôi về ý nghĩa.
    \item $r(e, t) \le 0$: Biến động có thể được giải thích hoàn toàn bằng nhiễu ngẫu nhiên do học lại.
\end{itemize}

\subsection{Chỉ số độ trôi chuẩn hóa ít bị ảnh hưởng bởi ngoại lai}
Phân phối nhiễu thực nghiệm thường lệch và có giá trị ngoại lai. Dùng độ lệch chuẩn mẫu để chuẩn hóa có thể làm méo thang đo. Vì vậy, nghiên cứu dùng độ lệch tuyệt đối trung vị MAD: lấy trung vị của khoảng cách tuyệt đối từ mỗi giá trị nhiễu đến trung vị chung:
\begin{equation}
\text{MAD}(\mathcal{R}_{-e}) = \text{median}\left( \left| n(u) - \text{median}(\mathcal{R}_{-e}) \right| \right), \quad u \in \mathcal{B}(e) \setminus \{e\}
\end{equation}
Với phân phối gần chuẩn, nhân MAD với hệ số hiệu chỉnh sẽ cho ước lượng độ phân tán ít bị ảnh hưởng bởi ngoại lai, tương đương độ lệch chuẩn:
\begin{equation}
\widehat{\sigma}_{\text{robust}} = 1{,}4826 \cdot \text{MAD}(\mathcal{R}_{-e})
\end{equation}

Chỉ số độ trôi chuẩn hóa một phía $\text{SED}^+$ được tính như sau:
\begin{equation}
\text{SED}^+(e, t) = \frac{\max\left(0, r(e, t)\right)}{1{,}4826 \cdot \text{MAD}\left(\mathcal{R}_{-e}\right) + \epsilon}
\label{eq:sed_plus}
\end{equation}
Trong đó, $\epsilon = 10^{-4}$ bảo vệ phép chia khi mẫu số gần 0, kết hợp với ngưỡng sàn $\text{MAD}_{\text{floor}} = 10^{-12}$.
Nói đơn giản, chỉ giữ phần dịch chuyển vượt nhiễu nền rồi chia cho độ dao động thông thường của nhiễu để có thang đo chung.

\begin{theorem}[Tính vững và đơn điệu của $\text{SED}^+$]
Chỉ số $\text{SED}^+(e, t)$ có các tính chất sau:
\begin{enumerate}
    \item \textbf{Không âm:} $\text{SED}^+(e, t) \ge 0, \forall e, t$.
    \item \textbf{Chỉ giữ phần vượt ngưỡng nhiễu:} $\text{SED}^+(e, t) \equiv 0$ khi và chỉ khi $r(e, t) \le 0$. Dao động không vượt trung vị nhiễu nền bị loại hoàn toàn, không tạo tín hiệu giả.
    \item \textbf{Không phụ thuộc thang độ phân tán:} Giá trị $\text{SED}^+$ đo phần vượt nhiễu bằng đơn vị độ lệch chuẩn ít bị ảnh hưởng bởi ngoại lai. Nhờ đó, có thể so sánh trực tiếp độ trôi giữa các nhóm bậc và giữa các mốc thời gian.
\end{enumerate}
\end{theorem}

\section{Suy luận nhiều bước và thước đo thứ hạng nghịch đảo}

\subsection{Bảng theo dõi câu hỏi nhiều bước}
Để kiểm tra độ trôi ảnh hưởng thế nào đến khả năng suy luận của hệ thống trí tuệ nhân tạo, nghiên cứu xây dựng một bảng theo dõi cố định gồm các câu hỏi nhiều bước, được đánh giá lặp lại theo thời gian. Mỗi câu hỏi $q \in \mathcal{Q}$ là một đường đi có hướng trên đồ thị:
\begin{equation}
q = \left(h, [r_1, r_2, \dots, r_k], ?t^*\right)
\end{equation}
Số chặng là $k \in \{1, 2\}$, cụ thể:
\begin{itemize}
    \item Với câu hỏi 1 chặng: $h \xrightarrow{r_1} t^*$.
    \item Với câu hỏi 2 chặng: $h \xrightarrow{r_1} e_{\text{inter}} \xrightarrow{r_2} t^*$.
\end{itemize}
Tập thực thể hỗ trợ trên đường suy luận của câu hỏi $q$ được ký hiệu là $\mathcal{S}_q = \{h, e_{\text{inter}}, t^*\}$. Câu hỏi bắt đầu từ chủ thể, đi qua các quan hệ và yêu cầu tìm đối tượng cuối cùng.

\subsection{Chấm điểm đáp án ứng viên bằng TransE nhiều bước}
Theo cách biểu diễn quan hệ bằng phép dịch chuyển của TransE, vector câu hỏi sau $k$ bước được tính bằng cách cộng lần lượt các vector quan hệ vào vector chủ thể:
\begin{equation}
\mathbf{v}_q = \mathbf{h} + \sum_{j=1}^k \mathbf{r}_j \in \mathbb{R}^d
\end{equation}
Với mỗi thực thể ứng viên $c \in \mathcal{C}$ trong tập tìm kiếm, lấy giá trị âm của khoảng cách Euclid $L_2$ làm điểm phù hợp:
\begin{equation}
S(q, c) = - \left\| \mathbf{v}_q - \mathbf{c} \right\|_2 = - \left\| \mathbf{h} + \sum_{j=1}^k \mathbf{r}_j - \mathbf{c} \right\|_2
\end{equation}
Điểm càng cao, tức càng gần 0, thì xác suất ứng viên $c$ là đáp án đúng càng lớn.

\subsection{Thứ hạng và thứ hạng nghịch đảo RR}
Thứ hạng của đáp án đúng $t^*$ trong tập ứng viên $\mathcal{C}$ được tính bằng:
\begin{equation}
\text{Rank}(q) = 1 + \sum_{c \in \mathcal{C} \setminus \{t^*\}} \mathbb{I}\left( S(q, c) > S(q, t^*) \right)
\end{equation}
Trong đó, $\mathbb{I}(\cdot)$ là hàm chỉ thị. Công thức đếm số ứng viên có điểm cao hơn đáp án đúng rồi cộng thêm một. Các ứng viên bằng điểm được xếp cùng hạng theo quy tắc xếp hạng thi đấu chuẩn.

Thứ hạng nghịch đảo RR của câu hỏi $q$ được định nghĩa như sau:
\begin{equation}
\text{RR}(q) = \begin{cases}
\displaystyle \frac{1}{\text{Rank}(q)}, & \text{nếu } t^* \in \mathcal{C}, \\[8pt]
0, & \text{nếu } t^* \notin \mathcal{C} \quad (\text{Hiện tượng Ngoài Vũ trụ -- OOV / OOU}).
\end{cases}
\label{eq:reciprocal_rank}
\end{equation}
RR là nghịch đảo thứ hạng nếu đáp án nằm trong tập ứng viên, và bằng không nếu đáp án nằm ngoài tập. MRR là trung bình RR của toàn bộ bảng câu hỏi tại một mốc thời gian:
\begin{equation}
\text{MRR} = \frac{1}{|\mathcal{Q}|} \sum_{q \in \mathcal{Q}} \text{RR}(q)
\end{equation}
Đáp án đúng được xếp càng cao thì RR càng lớn; MRR tóm tắt chất lượng xếp hạng của cả bảng câu hỏi.
\subsection{Các nhánh đối chiếu và tập đáp án ứng viên}
Để tách tác động riêng của độ trôi khỏi các yếu tố gây nhiễu khi dữ liệu thay đổi, nghiên cứu dùng các nhánh đối chiếu:

\begin{enumerate}
    \item \textbf{Arm A -- Nhánh giữ nguyên mô hình T01:} Mô hình biểu diễn vector được giữ cố định ở trạng thái đã huấn luyện tại $T_{01}$, không cập nhật trọng số.
    \item \textbf{Arm B -- Nhánh cập nhật tại $T_t$:} Mô hình biểu diễn vector được huấn luyện lại hoàn toàn tại mỗi mốc $T_t$ khi có tri thức mới.
    \item \textbf{Arm C -- Nhánh hỗ trợ bằng đường đi:} Bổ sung đặc trưng từ các đường đi có bằng chứng ngữ nghĩa để hỗ trợ xếp hạng khi chất lượng biểu diễn vector giảm.
\end{enumerate}

Đồng thời, nghiên cứu đánh giá trên hai tập đáp án ứng viên:
\begin{itemize}
    \item \textbf{Tập thực thể đã có tại T1 $\mathcal{C} = \mathcal{E}_{T01}$:} Chỉ xét các thực thể đã xuất hiện ở mốc ban đầu. Cách này loại $100\%$ ảnh hưởng từ việc tăng số ứng viên hoặc thêm thực thể mới, để chỉ đo tác động của độ trôi biểu diễn.
    \item \textbf{Tập mở $\mathcal{C} = \mathcal{E}_{T_t}$:} Tập tìm kiếm mở rộng theo sự phát triển của đồ thị thực tế.
\end{itemize}

\section{Mô hình hiệu ứng cố định hai chiều TWFE}
\label{sec:twfe_econometrics}

\subsection{Mô hình bảng theo dõi và giả thuyết \texorpdfstring{$H_1$}{H1}}
Để kiểm định giả thuyết về tác động nhân quả của độ trôi lên chất lượng suy luận nhiều bước, nghiên cứu dùng mô hình hiệu ứng cố định hai chiều TWFE. Đây là phương pháp thông dụng trong kinh tế lượng ứng dụng để kiểm soát các yếu tố không quan sát được:

\begin{hypothesis}[Giả thuyết $H_1$ -- Độ trôi biểu diễn làm giảm chất lượng suy luận]
Khi độ trôi biểu diễn đã chuẩn hóa ($\text{SED}^+$) của các thực thể hỗ trợ tăng, thứ hạng trả lời câu hỏi nhiều bước giảm một cách có ý nghĩa thống kê ($\beta_d < 0$).
\end{hypothesis}

Mô hình dùng bảng theo dõi cân bằng gồm $N = 144$ câu hỏi. Mỗi câu hỏi được đánh giá qua $T = 9$ chuyển tiếp thời gian, tạo thành tổng số quan sát $M = N \times T = 1,296$:
\begin{equation}
\text{RR}_{i,t} = \alpha_i + \gamma_t + \beta_d \cdot \overline{\text{SED}}^+_{i,t} + \varepsilon_{i,t}
\label{eq:twfe_specification}
\end{equation}
trong đó:
\begin{itemize}
    \item $\text{RR}_{i,t}$ là thứ hạng nghịch đảo của câu hỏi $i$ tại chuyển tiếp $t$.
    \item $\alpha_i$ là hiệu ứng cố định của câu hỏi. Nó bao gồm mọi đặc điểm không đổi theo thời gian, như độ phức tạp ý nghĩa, số chặng và mức đa nghĩa của thực thể đầu.
    \item $\gamma_t$ là hiệu ứng cố định của chuyển tiếp thời gian. Nó kiểm soát toàn bộ biến động chung của đồ thị tại $t$, như tốc độ mở rộng, số cạnh mới và mật độ liên kết.
    \item $\overline{\text{SED}}^+_{i,t} = \text{median}_{e \in \mathcal{S}_i}\left(\text{SED}^+(e, t)\right)$ là trung vị độ trôi đã chuẩn hóa của các thực thể hỗ trợ câu hỏi $i$.
    \item $\beta_d$ là hệ số cần ước lượng, cho biết mối liên hệ giữa độ trôi và RR.
    \item $\varepsilon_{i,t}$ là sai số ngẫu nhiên, thỏa mãn điều kiện ngoại sinh nghiêm ngặt sau khi kiểm soát $\alpha_i$ và $\gamma_t$.
\end{itemize}
Nói đơn giản, mô hình tách độ khó vốn có của từng câu hỏi và biến động chung của từng thời điểm trước khi xem độ trôi đi cùng thay đổi chất lượng trả lời ra sao.
\subsection{Loại hiệu ứng cố định bằng cách trừ trung bình hai chiều}
Để ước lượng $\beta_d$ không bị lệch do các yếu tố $\alpha_i$ và $\gamma_t$, ta trừ trung bình theo cả câu hỏi lẫn thời gian.
Các giá trị trung bình mẫu gồm trung bình theo câu hỏi, theo thời điểm và trên toàn bảng:
\begin{equation}
\overline{y}_{i\cdot} = \frac{1}{T} \sum_{t=1}^T y_{it}, \quad \overline{y}_{\cdot t} = \frac{1}{N} \sum_{i=1}^N y_{it}, \quad \overline{y}_{\cdot\cdot} = \frac{1}{NT} \sum_{i=1}^N \sum_{t=1}^T y_{it}
\end{equation}
Tính tương tự cho biến giải thích $\overline{x}_{i\cdot}, \overline{x}_{\cdot t}, \overline{x}_{\cdot\cdot}$. Sau đó, trừ trung bình của câu hỏi và thời điểm, rồi cộng lại trung bình chung:
\begin{equation}
\widetilde{y}_{it} = y_{it} - \overline{y}_{i\cdot} - \overline{y}_{\cdot t} + \overline{y}_{\cdot\cdot}
\end{equation}
\begin{equation}
\widetilde{x}_{it} = x_{it} - \overline{x}_{i\cdot} - \overline{x}_{\cdot t} + \overline{x}_{\cdot\cdot}
\end{equation}
Khi đó, phương trình \eqref{eq:twfe_specification} trở thành hồi quy tuyến tính không có hệ số chặn:
\begin{equation}
\widetilde{y}_{it} = \beta_d \cdot \widetilde{x}_{it} + \widetilde{\varepsilon}_{it}
\end{equation}
Ước lượng bình phương tối thiểu cho $\beta_d$ là:
\begin{equation}
\widehat{\beta}_d = \left( \widetilde{\mathbf{x}}^T \widetilde{\mathbf{x}} \right)^{-1} \widetilde{\mathbf{x}}^T \widetilde{\mathbf{y}} = \frac{\sum_{i=1}^N \sum_{t=1}^T \widetilde{x}_{it} \widetilde{y}_{it}}{\sum_{i=1}^N \sum_{t=1}^T \widetilde{x}_{it}^2}
\label{eq:ols_demeaned}
\end{equation}
Công thức này lấy tổng tích của hai biến sau khi trừ trung bình, chia cho tổng bình phương của biến giải thích đã biến đổi.
\subsection{Sai số chuẩn theo cụm chuyển tiếp thời gian}
Các câu hỏi trong cùng chuyển tiếp $t$ dùng chung không gian vector và cùng chịu biến động ngẫu nhiên khi huấn luyện. Vì vậy, sai số $\varepsilon_{i,t}$ có tương quan trong cùng cụm. Giả định các sai số độc lập và cùng phân phối bị vi phạm nghiêm trọng.

Nghiên cứu xử lý bằng ma trận hiệp phương sai có xét sự phụ thuộc trong từng cụm chuyển tiếp:
\begin{equation}
\widehat{\text{Var}}_{\text{cluster}}(\widehat{\beta}_d) = \left(\frac{G}{G-1}\right) \left(\frac{M-1}{M-K}\right) \left( \widetilde{\mathbf{x}}^T \widetilde{\mathbf{x}} \right)^{-1} \left( \sum_{g=1}^G \mathbf{s}_g \mathbf{s}_g^T \right) \left( \widetilde{\mathbf{x}}^T \widetilde{\mathbf{x}} \right)^{-1}
\label{eq:clustered_se}
\end{equation}
trong đó:
\begin{itemize}
    \item $G = 9$ là tổng số cụm, ứng với 9 chuyển tiếp thời gian.
    \item $M = 1,296$ là tổng số quan sát; $K = 1$ là số biến hồi quy.
    \item $\mathbf{s}_g = \sum_{i \in \text{cụm } g} \widetilde{x}_{ig} \widehat{\varepsilon}_{ig}$ là vector điểm số của cụm $g$, tổng hợp tích giữa biến giải thích và phần dư trong cụm.
\end{itemize}
Công thức phương sai cộng đóng góp của từng cụm, rồi hiệu chỉnh theo số cụm, số quan sát và số biến hồi quy.
\subsection{Suy luận với ít cụm bằng bootstrap theo cụm, trọng số Webb}
Công thức \eqref{eq:clustered_se} đã xét tương quan trong cụm, nhưng lý thuyết của nó đòi hỏi số cụm lớn, tức $G \to \infty$, thường là $G \ge 30 \sim 50$. Khi chỉ có $G = 9$ chuyển tiếp, sai số chuẩn theo cụm bị đánh giá thấp nghiêm trọng. Điều này làm tăng sai lầm loại I: bác bỏ nhầm giả thuyết $H_0$ có thể lên tới $20-40\%$ dù mức ý nghĩa đặt ở $5\%$.

Để xử lý số cụm rất ít, nghiên cứu dùng \textbf{bootstrap theo cụm có ràng buộc giả thuyết không, dựa trên thống kê t} \cite{cameron2008bootstrap, mackinnon2017wild}, với phân phối trọng số Webb 6 điểm \cite{webb2014reworking}.

\begin{definition}[Phân phối trọng số Webb 6 điểm]
Phân phối gán xác suất $1/6$ cho mỗi giá trị trong 6 giá trị căn bậc hai dưới đây:
\begin{equation}
v_g \sim \mathcal{W}_6 = \left\{ -\sqrt{\frac{3}{2}}, -1, -\sqrt{\frac{1}{2}}, \sqrt{\frac{1}{2}}, 1, \sqrt{\frac{3}{2}} \right\}
\end{equation}
Trọng số Webb có các mô-men $\mathbb{E}[v_g] = 0$, $\mathbb{E}[v_g^2] = 1$, $\mathbb{E}[v_g^3] = 0$ và $\mathbb{E}[v_g^4] = 7/6$. Phân phối này tối ưu hơn hẳn phân phối Rademacher hai điểm ($\{\pm 1\}$) khi số cụm $G < 10$. Lý do là Rademacher chỉ tạo được $2^{G-1} = 2^8 = 256$ tổ hợp dấu độc lập, chưa đủ để có phân phối mẫu bootstrap phong phú.
\end{definition}

\begin{algorithm}[H]
\caption{Kiểm định Giả thuyết $H_0: \beta_d = 0$ bằng bootstrap theo cụm với trọng số Webb}
\label{alg:wcb}
\begin{algorithmic}[1]
\Require Dữ liệu đã khử trung bình $(\widetilde{\mathbf{y}}, \widetilde{\mathbf{x}})$, chỉ số cụm thời gian $\{g\}$, số lần bốc mẫu $B = 9999$, seed $= 101$.
\State Ước lượng mô hình ban đầu bằng OLS: tính $\widehat{\beta}_d$, phần dư $\widehat{\varepsilon}$, và giá trị kiểm định thực tế $t_{\text{orig}} = |\widehat{\beta}_d / \text{SE}_{\text{cluster}}(\widehat{\beta}_d)|$.
\State Khởi tạo bộ đếm cực biên: $C_{\text{extreme}} = 0$.
\For{$b = 1 \to B$}
    \State Lấy ngẫu nhiên trọng số Webb cho từng cụm: $v_g^{(b)} \sim \text{Uniform}(\mathcal{W}_6)$ cho mỗi $g \in \{1, \dots, G\}$.
    \State Gán trọng số cụm cho từng quan sát: $w_{it} = v_{g(t)}^{(b)}$.
    \State Dưới giả thuyết vô hiệu $H_0: \beta_d = 0$, sinh biến phụ thuộc giả lập:
    \begin{equation*}
    y_{it}^* = w_{it} \cdot \widehat{\varepsilon}_{it}
    \end{equation*}
    \State Ước lượng lại OLS trên dữ liệu giả lập: $\widehat{\beta}_d^* = (\widetilde{\mathbf{x}}^T \widetilde{\mathbf{x}})^{-1} \widetilde{\mathbf{x}}^T \mathbf{y}^*$.
    \State Tính toán phần dư bootstrap: $\varepsilon^* = \mathbf{y}^* - \widetilde{\mathbf{x}} \widehat{\beta}_d^*$.
    \State Tính sai số chuẩn phân cụm bootstrap $\text{SE}_{\text{cluster}}^*(\widehat{\beta}_d^*)$ theo công thức \eqref{eq:clustered_se}.
    \State Tính đại lượng kiểm định bootstrap: $t_b^* = |\widehat{\beta}_d^* / \text{SE}_{\text{cluster}}^*(\widehat{\beta}_d^*)|$.
    \If{$t_b^* \ge t_{\text{orig}}$}
        \State $C_{\text{extreme}} \leftarrow C_{\text{extreme}} + 1$.
    \EndIf
\EndFor
\State Tính toán p-value bootstrapping hiệu chỉnh hữu hạn:
\begin{equation}
p_{\text{WCB}} = \frac{C_{\text{extreme}} + 1}{B + 1}
\end{equation}
\State \Return $p_{\text{WCB}}$ và thống kê kiểm định.
\end{algorithmic}
\end{algorithm}
Nói đơn giản, bootstrap tạo lại dữ liệu nhiều lần bằng cách đổi trọng số của cả cụm, rồi xem kết quả thật có khác thường so với các kết quả tạo lại hay không. Giá trị xác suất kiểm định là tỷ lệ số lần thống kê tạo lại không nhỏ hơn thống kê thực tế, có hiệu chỉnh cho số lần lấy mẫu hữu hạn.

\section{Phép kiểm chứng bằng độ trôi giả}
\label{sec:falsification_pseudo_drift}

\subsection{Dùng đối chứng để phát hiện tương quan giả}
Trong kinh tế lượng và suy luận nhân quả, tương quan giữa biến tác động $\overline{\text{SED}}^+_{i,t}$ và biến kết quả $\text{RR}_{i,t}$ có thể chỉ là tương quan giả. Những nguyên nhân có thể gồm:
\begin{itemize}
    \item Biến động ngẫu nhiên của thuật toán tối ưu.
    \item Quá khớp cục bộ do các phép biến đổi phi tuyến như $\max(0, \cdot)$.
    \item Hai chuỗi thời gian độc lập tình cờ biến động cùng chiều.
\end{itemize}
Để bảo đảm kết luận khoa học có tính hợp lệ vô điều kiện, nghiên cứu dùng một biến đối chứng âm để kiểm tra khả năng xuất hiện tương quan giả.

\subsection{Biến độ trôi giả (\texorpdfstring{$D_{\text{pseudo}}$}{D\_pseudo})}
Nghiên cứu tạo biến ngẫu nhiên \textbf{độ trôi giả} ($D_{\text{pseudo}}$) hoàn toàn từ nhiễu trắng Gauss độc lập. Biến này được chuẩn hóa theo đúng quy trình và có cùng khoảng giá trị hình học với độ trôi thật $\text{SED}^+$:
\begin{equation}
D_{\text{pseudo}}(i, t) = \frac{\max\left(0, \eta_{it}\right)}{1{,}4826 \cdot \overline{\text{MAD}}_{\text{null}} + \epsilon}
\label{eq:pseudo_drift_def}
\end{equation}
trong đó:
\begin{itemize}
    \item $\eta_{it} \sim \mathcal{N}(0, \sigma^2)$ là biến Gauss độc lập, có mức biến thiên tương đương phương sai nền của đồ thị: $\sigma = \text{median}_{e \in \mathcal{S}_i}(\text{MAD}(\mathcal{R}_{-e}))$.
    \item Mẫu số $1{,}4826 \cdot \overline{\text{MAD}}_{\text{null}} + \epsilon$ dùng đúng cách chuẩn hóa ít bị ảnh hưởng bởi ngoại lai trong \eqref{eq:sed_plus}.
\end{itemize}

Biến $D_{\text{pseudo}}$ có cùng miền giá trị $[0, +\infty)$, cùng tập trung xác suất tại 0 do phép $\max(0, \cdot)$ và cùng bậc phương sai với $\text{SED}^+$. Tuy nhiên, $D_{\text{pseudo}}$ không chứa thông tin nào về sự dịch chuyển thật của vector thực thể trong đồ thị tri thức.

\subsection{Hồi quy đối chứng và điều kiện chấp nhận kết quả}
Xây dựng thêm một mô hình TWFE song song với mô hình chính, thay độ trôi thật bằng độ trôi giả:
\begin{equation}
\text{RR}_{i,t} = \alpha_i + \gamma_t + \beta_{\text{pseudo}} \cdot D_{\text{pseudo}}(i,t) + \varepsilon_{i,t}'
\label{eq:falsification_twfe}
\end{equation}
Mô hình đối chứng dùng cùng cách trừ trung bình hai chiều. Giả thuyết $H_0: \beta_{\text{pseudo}} = 0$ được kiểm định bằng bootstrap theo cụm với trọng số Webb 6 điểm ($B = 9.999$ trong phiên bản thực hiện lại).


\begin{criterion}[Điều kiện kiểm tra tính hợp lệ thống kê và tương quan giả]
Khung lý thuyết và mô hình kinh tế lượng đặt ra hai điều kiện chấp nhận kết quả thực nghiệm:
\begin{enumerate}
    \item \textbf{Quan hệ theo lý thuyết và kiểm định theo cụm ($H_1$):}
    \begin{equation}
    \widehat{\beta}_d < 0
    \end{equation}
    Ước lượng $\widehat{\beta}_d < 0$ thể hiện quan hệ ngược chiều mà lý thuyết dự đoán giữa độ trôi và khả năng suy luận nhiều bước. Khi chỉ có ít cụm thời gian ($G = 9$ cụm), giá trị $p$ từ bootstrap theo cụm nhỏ ($p_{\text{WCB}}$) cho phép kiểm định chính xác liệu quan hệ này có khác 0 sau khi đã xét biến động chung của từng cụm thời gian hay không.
    \item \textbf{Phép kiểm chứng bằng độ trôi giả:}
    \begin{equation}
    \widehat{\beta}_{\text{pseudo}} \approx 0 \quad \text{và} \quad p_{\text{WCB}}(\widehat{\beta}_{\text{pseudo}}) > 0{,}05
    \end{equation}
    Phép kiểm bằng độ trôi giả đạt khi $p_{\text{WCB}}(\widehat{\beta}_{\text{pseudo}}) > 0{,}05$. Đây là điều kiện \textit{cần}, không phải \textit{đủ}. Nó cho thấy riêng nhiễu do học lại không tạo liên hệ giả với kết quả. Muốn biết phép đo có đủ nhạy để phát hiện độ trôi thật hay không, vẫn cần phép thử đối chứng dương.
\end{enumerate}
\end{criterion}
Nói đơn giản, phép kiểm bằng độ trôi giả giúp xem quy trình có nhầm nhiễu thành một mối liên hệ có ý nghĩa hay không.
Vì vậy, phải dùng phép kiểm bằng độ trôi giả cùng phép thử đối chứng dương, tức bơm độ trôi có chủ đích, và phân tích độ nhạy. Chỉ khi cả ba cùng đạt mới có thể diễn giải hệ số $\beta_d$, dù khác 0 hay bằng 0, như một kết luận về tác động của độ trôi biểu diễn.
